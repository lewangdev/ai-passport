cmake_minimum_required(VERSION 3.20)
project(toolkit_render C CXX)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(toolkit_render render.c ../../main/toolkit_model.c ../../main/toolkit_view.c ../../assets/fonts/toolkit_font_16.c)
target_include_directories(toolkit_render PRIVATE ../../main)
target_compile_definitions(toolkit_render PRIVATE TK_RENDER_TEST)
target_compile_options(toolkit_render PRIVATE -UNDEBUG)
target_link_libraries(toolkit_render PRIVATE lvgl m)
